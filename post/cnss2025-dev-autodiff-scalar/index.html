Automatic_Differentiation_For_Scalar

30% 能对标量函数实现 自动微分，标量函数为基本初等函数的复合函数。也就是说，你的自动微分程序中需要包含所有基本初等函数的求导

因为后面跟这部分是相关的，所以这部分就写详细点吧qwq

  • 自动微分（Automatic Differentiation）

    • 自动微分 其实就是利用 链式求导法则，在进行 传播 的时候计算 微分值，这个过程可以在 计算图 中体现，这样每次利用的就是 已经求过 的值，就可以把 复合函数 转为 基本初等函数 解决了

    • 自动微分 分为 正向模式 和 反向模式，这里我就直接用 反向模式 了，因为一般来说 输入维度 大于 输出维度。然后可以结合 反向传播 来做

  • 反向传播

    • 反向传播 就是利用 正向传播 后算出来的 误差，来计算出各个 参数 的 梯度，然后通过 梯度下降法，得到新的 参数，以此来达到 训练 的作用

      当然在 自动微分 的过程中不需要去 改变参数，因为只需要求得对 对应点的梯度，所以代码中也不用体现 改变参数 这个过程

  • 代码

    • 这里的话我就参考 pytorch 来实现了，然后可以用 pytorch 作为测试（所以导入的 库 中有 pytorch 但我实现上是没用的，仅用来测试），最后可以看看 手写的 和 pytorch 的 自动微分 之间的 差异

    • 张量类

      • 这里先 定义 一个 张量类，第一问 这里其实是只有 标量 了，然后其实现了 反向传播 和 自动微分

      • 反向传播

        • 反向传播 的话需要知道 节点的拓扑序，这里我就直接用一个 拓扑排序 来做了。然后 拓扑排序 的话可以用 dfs 和 bfs

          由于 bfs 要维护的 数据 稍微有点多，比如 入度 之类的，我这里就直接用 dfs 吧，反正 复杂度 是一样的，无非就是 递归 的常数大了一点

#反向传播

def backward(self):

    #拓扑排序

    topo = []

    visited = set()

    #由于是反向的,这里直接用dfs

    def build_topo(v):

        if v not in visited:

            visited.add(v)

            for parent in v._prev:

                build_topo(parent)

            topo.append(v)



    build_topo(self)



    #反向传播

    #初始化,根节点求导是1

    self.grad = np.ones_like(self.data)



    #反拓扑序

    for v in reversed(topo):

        v._backwrad()

        • 其中这个 _backward 会在后续对 基本初等函数 求导的过程被赋值

  # 加

    def __add__(self, other):

        # 确保 other 是张量类

        other = other if isinstance(other, Tensor) else Tensor(other, requires_grad = False)

        # 创建 self 和 other 的儿子

        out = Tensor(self.data + other.data, (self, other), '+')



        # 反向传播函数

        def _backward():

            # a = b + c



            # dL/db = dL/da * da/db

            # d(b+c)/db = 1

            if self.requires_grad:

                self.grad += 1 * out.grad

            # dL/dc = dL/da * da/dc

            # d(b+c)/dc = 1

            if other.requires_grad:

                other.grad += 1 * out.grad



        out._backward = _backward



        return out

    # 反向加

    def __radd__(self, other):

        return self + other

    def __mul__(self, other):

        # 确保 other 是张量类

        other = other if isinstance(other, Tensor) else Tensor(other, requires_grad = False)

        # 创建 self 和 other 的儿子

            # a = b * c

            # d(a*b)/da = b

                self.grad += other.data * out.grad

                other.grad += self.data * out.grad

        return out

  # 幂运算

    def __pow__(self, other):

        # x^y

        other = other if isinstance(other, Tensor) else Tensor(other, requires_grad = False)

        out = Tensor(self.data ** other.data, (self, other), '**')



        def _backward():

            # d(x^y)/dx = y * x^(y-1)

            if self.data != 0: 

                # 防止 0^负数

                if self.requires_grad:

                    self.grad += other.data * (self.data ** (other.data - 1)) * out.grad

            # d(x^y)/∂y = x^y * ln(x)

            if self.data > 0:

                # 防止 ln(负数) 或 ln(0)

                if other.requires_grad:

                    other.grad += (self.data ** other.data) * np.log(self.data) * out.grad



        out._backward = _backward



        return out

    # 反向幂运算,即指数函数

    def __rpow__(self, other):

        # a^x

        # other^self，other 是常数，self 是变量



        # 底数要大于0

        assert other > 0, "base must be positive for general exponential"



        out = Tensor(other ** self.data, (self,), f'{other}**x')



        def _backward():

            # d(a^x)/dx = a^x * ln(a)

            if self.requires_grad:

                self.grad += (other ** self.data) * np.log(other) * out.grad



        out._backward = _backward



        return out

    def exp(self):

        # e^x

        def _backward():

            # d(e^x)/dx = e^x

            if self.requires_grad:

                self.grad += np.exp(self.data) * out.grad



        out._backward = _backward

        return out

    def log_a(self, a):        

        # 底数和真数检查

        assert np.all(self.data > 0), f"log_{a} undefined for non-positive values"

        assert a > 0 and a != 1, "a must be positive and not equal to 1"



        out = Tensor(np.log(self.data) / np.log(a), (self,), f'log_{a}')



        def _backward():

            # d(log_a(x))/dx = 1 / (x * ln(a))

            if self.requires_grad:

                self.grad += (1.0 / (self.data * np.log(a))) * out.grad



        out._backward = _backward



        return out

    def log(self):

        # ln(x)



        # 确保真数大于0

        assert np.all(self.data > 0), "log undefined for non-positive values"



        out = Tensor(np.log(self.data), (self,), 'log')



        def _backward():

            # d(ln(x))/dx = 1/x

            if self.requires_grad:

                self.grad += (1.0 / self.data) * out.grad



        out._backward = _backward

        return out

    # sec

    def sec(self):

        # sec(x)

        out = Tensor(1.0 / np.cos(self.data), (self,), 'sec')



        def _backward():

            # d(sec(x))/dx = tan(x)/cos(x)

            sec_val = 1.0 / np.cos(self.data)

            tan_val = np.tan(self.data)

            if self.requires_grad:

        out._backward = _backward



        return out



    # csc

    def csc(self):

        # csc(x)

        out = Tensor(1.0 / np.sin(self.data), (self,), 'csc')



        def _backward():

            # d(sec(x))/dx = tan(x)/sin(x)

            csc_val = 1.0 / np.sin(self.data)

            cot_val = 1.0 / np.tan(self.data)

            if self.requires_grad:

                self.grad += (-csc_val * cot_val) * out.grad



        out._backward = _backward



        return out

  # 反三角函数

    # arcsin

    def arcsin(self):

        # arcsin(x)



        # 判断x是否在[-1,1]

        assert np.all(self.data >= -1) and np.all(self.data <= 1), "arcsin undefined outside [-1, 1]"



        out = Tensor(np.arcsin(self.data), (self,), 'arcsin')



        def _backward():

            # d(arcsin(x))/dx = 1/sqrt(1-x^2)

            if self.requires_grad:

                self.grad += (1.0 / np.sqrt(1 - self.data**2)) * out.grad



        out._backward = _backward



        return out



    # arccos

    def arccos(self):

        # arccos(x)



        # 判断x是否在[-1,1]

        assert np.all(self.data >= -1) and np.all(self.data <= 1), "arccos undefined outside [-1, 1]"



        out = Tensor(np.arccos(self.data), (self,), 'arccos')



        def _backward():

            # d(arccos(x))/dx = -1/sqrt(1-x^2)

            if self.requires_grad:

                self.grad += (-1.0 / np.sqrt(1 - self.data**2)) * out.grad



        out._backward = _backward



        return out



    # arctan

    def arctan(self):

        # arctan(x)

        out = Tensor(np.arctan(self.data), (self,), 'arctan')



        def _backward():

            # d(arctan(x))/dx = 1/1+x^2

            if self.requires_grad:

                self.grad += (1.0 / (1 + self.data**2)) * out.grad



        out._backward = _backward



        return out

import numpy as np

import torch



class Tensor:

    """

    张量类,实现自动微分

    """

    def __init__(self, data, __parents = (), _op = '', lable = '', requires_grad = True):

        # 确保 data 是 np 数组

        if not isinstance(data, np.ndarray):

            data = np.array(data, dtype=np.float64)



        self.data = data

        self.grad = np.zeros_like(self.data) # 梯度初始化为 0



        # 反向传播函数 

        self._backward = lambda: None



        # 记录当前 Tensor 的父节点

        self._prev = set(__parents)



        # 调试

        self._op = _op

        self.lable = lable



        # 是否需要计算梯度

        self.requires_grad = requires_grad



    # 调试

    def __repr__(self):

        return f"Tensor(data = {self.data}, grad = {self.grad})"



    # 反向传播

    def backward(self):

        # 拓扑排序

        topo = []

        visited = set()

        # 由于是反向的,这里直接用dfs

        def build_topo(v):

            if v not in visited:

                visited.add(v)

                for parent in v._prev:

                    build_topo(parent)

                topo.append(v)



        build_topo(self)



        # 反向传播

        # 初始化,根节点求导是1

        self.grad = np.ones_like(self.data)



        # 反拓扑序

        for v in reversed(topo):

            v._backward()



    # 算术运算

    # 加

    def __add__(self, other):

        # 确保 other 是张量类

        other = other if isinstance(other, Tensor) else Tensor(other, requires_grad = False)

        # 创建 self 和 other 的儿子

        out = Tensor(self.data + other.data, (self, other), '+')



        # 反向传播函数

        def _backward():

            # a = b + c



            # dL/db = dL/da * da/db

            # d(b+c)/db = 1

            if self.requires_grad:

                self.grad += 1 * out.grad

            # dL/dc = dL/da * da/dc

            # d(b+c)/dc = 1

            if other.requires_grad:

                other.grad += 1 * out.grad



        out._backward = _backward



        return out

    # 反向加

    def __radd__(self, other):

        return self + other



    # 取负

    def __neg__(self):

        return self * -1



    # 减

    def __sub__(self, other):

        # a = b - c = b + (-c)

        return self + (-other)

    # 反向减

    def __rsub__(self, other):

        return other + (-self)



    # 乘

    def __mul__(self, other):

        # 确保 other 是张量类

        other = other if isinstance(other, Tensor) else Tensor(other, requires_grad = False)

        # 创建 self 和 other 的儿子

        out = Tensor(self.data * other.data, (self, other), '*')



        # 反向传播函数

        def _backward():

            # a = b * c

            # dL/db = dL/da * da/db

            # d(a*b)/da = b

            if self.requires_grad:

                self.grad += other.data * out.grad

            # dL/dc = dL/da * da/dc

            # d(a*b)/db = a

            if other.requires_grad:

                other.grad += self.data * out.grad



        out._backward = _backward



        return out

    # 反向乘

    def __rmul__(self, other):

        return self * other



    # 除

    def __truediv__(self, other):

        return self * other**-1

    # 反向除

    def __rtruediv__(self, other):

        return other * self**-1



    # 幂运算

    def __pow__(self, other):

        # x^y

        other = other if isinstance(other, Tensor) else Tensor(other, requires_grad = False)

        out = Tensor(self.data ** other.data, (self, other), '**')



        def _backward():

            # d(x^y)/dx = y * x^(y-1)

            if self.data != 0: 

                # 防止 0^负数

                if self.requires_grad:

                    self.grad += other.data * (self.data ** (other.data - 1)) * out.grad

            # d(x^y)/∂y = x^y * ln(x)

            if self.data > 0:

                # 防止 ln(负数) 或 ln(0)

                if other.requires_grad:

                    other.grad += (self.data ** other.data) * np.log(self.data) * out.grad



        out._backward = _backward



        return out

    # 反向幂运算,即指数函数

    def __rpow__(self, other):

        # a^x

        # other^self，other 是常数，self 是变量



        # 底数要大于0

        assert other > 0, "base must be positive for general exponential"



        out = Tensor(other ** self.data, (self,), f'{other}**x')



        def _backward():

            # d(a^x)/dx = a^x * ln(a)

            if self.requires_grad:

                self.grad += (other ** self.data) * np.log(other) * out.grad



        out._backward = _backward



        return out



    # e^x

    def exp(self):

        # e^x

        out = Tensor(np.exp(self.data), (self,), 'exp')



        def _backward():

            # d(e^x)/dx = e^x

            if self.requires_grad:

                self.grad += np.exp(self.data) * out.grad



        out._backward = _backward

        return out



    # log_a(x)

    def log_a(self, a):        

        # 底数和真数检查

        assert np.all(self.data > 0), f"log_{a} undefined for non-positive values"

        assert a > 0 and a != 1, "a must be positive and not equal to 1"



        out = Tensor(np.log(self.data) / np.log(a), (self,), f'log_{a}')



        def _backward():

            # d(log_a(x))/dx = 1 / (x * ln(a))

            if self.requires_grad:

                self.grad += (1.0 / (self.data * np.log(a))) * out.grad



        out._backward = _backward



        return out



    # ln

    def log(self):

        # ln(x)



        # 确保真数大于0

        assert np.all(self.data > 0), "log undefined for non-positive values"



        out = Tensor(np.log(self.data), (self,), 'log')



        def _backward():

            # d(ln(x))/dx = 1/x

            if self.requires_grad:

                self.grad += (1.0 / self.data) * out.grad



        out._backward = _backward

        return out



    # 三角函数

    # sin

    def sin(self):

        # sin(x)

        out = Tensor(np.sin(self.data), (self,), 'sin')



        def _backward():

            # d(sin(x))/dx = cos(x)

            if self.requires_grad:

                self.grad += np.cos(self.data) * out.grad



        out._backward = _backward



        return out



    # cos

    def cos(self):

        # cos(x)

        out = Tensor(np.cos(self.data), (self,), 'cos')



        def _backward():

            # d(cos(x))/dx = -sin(x)

            if self.requires_grad:

                self.grad += -np.sin(self.data) * out.grad

        out._backward = _backward

        return out



    # tan

    def tan(self):

        # tan(x)

        out = Tensor(np.tan(self.data), (self,), 'tan')



        def _backward():

            # d(tan(x))/dx = 1/(cos(x)^2)

            if self.requires_grad:

                self.grad += (1.0 / (np.cos(self.data) ** 2)) * out.grad

        out._backward = _backward

        return out



    # cot

    def cot(self):

        # cot(x)

        out = Tensor(1.0 / np.tan(self.data), (self,), 'cot')



        def _backward():

            # d(cot(x))/dx = -1/(sin(x)^2)

            if self.requires_grad:

                self.grad += (-1.0 / (np.sin(self.data) ** 2)) * out.grad



        out._backward = _backward



        return out



    # sec

    def sec(self):

        # sec(x)

        out = Tensor(1.0 / np.cos(self.data), (self,), 'sec')



        def _backward():

            # d(sec(x))/dx = tan(x)/cos(x)

            sec_val = 1.0 / np.cos(self.data)

            tan_val = np.tan(self.data)

            if self.requires_grad:

                self.grad += (sec_val * tan_val) * out.grad



        out._backward = _backward



        return out



    # csc

    def csc(self):

        # csc(x)

        out = Tensor(1.0 / np.sin(self.data), (self,), 'csc')



        def _backward():

            # d(sec(x))/dx = tan(x)/sin(x)

            csc_val = 1.0 / np.sin(self.data)

            cot_val = 1.0 / np.tan(self.data)

            if self.requires_grad:

                self.grad += (-csc_val * cot_val) * out.grad



        out._backward = _backward



        return out



    # 反三角函数

    # arcsin

    def arcsin(self):

        # arcsin(x)



        # 判断x是否在[-1,1]

        assert np.all(self.data >= -1) and np.all(self.data <= 1), "arcsin undefined outside [-1, 1]"



        out = Tensor(np.arcsin(self.data), (self,), 'arcsin')



        def _backward():

            # d(arcsin(x))/dx = 1/sqrt(1-x^2)

            if self.requires_grad:

                self.grad += (1.0 / np.sqrt(1 - self.data**2)) * out.grad



        out._backward = _backward



        return out



    # arccos

    def arccos(self):

        # arccos(x)



        # 判断x是否在[-1,1]

        assert np.all(self.data >= -1) and np.all(self.data <= 1), "arccos undefined outside [-1, 1]"



        out = Tensor(np.arccos(self.data), (self,), 'arccos')



        def _backward():

            # d(arccos(x))/dx = -1/sqrt(1-x^2)

            if self.requires_grad:

                self.grad += (-1.0 / np.sqrt(1 - self.data**2)) * out.grad



        out._backward = _backward



        return out



    # arctan

    def arctan(self):

        # arctan(x)

        out = Tensor(np.arctan(self.data), (self,), 'arctan')



        def _backward():

            # d(arctan(x))/dx = 1/1+x^2

            if self.requires_grad:

                self.grad += (1.0 / (1 + self.data**2)) * out.grad



        out._backward = _backward



        return out
