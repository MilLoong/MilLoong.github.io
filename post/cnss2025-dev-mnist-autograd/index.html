MNIST 数据集

  • 训练

    • 思路就是在每个 epoch 里去处理 数据集的每个批次

    • 评估训练准确度 用的是 训练集的准确度 $train\_acc = \frac{train\_correct}{ train\_total}$ 和 测试集的准确度 $test\_acc = evaluate\_accuracy(model, test\_loader)$，其中 $evaluate\_accuracy()$ 是：

      def evaluate_accuracy(model, data_loader):

    # 在给定的数据加载器上评估模型准确率

    correct = 0

    total = 0


    for data, target in data_loader:

        # 将 PyTorch 的 tensor 转为 numpy 数组并展开为向量

        x_np = data.view(data.shape[0], -1).numpy()


        # 原始输出

        logits = forward_np_batch(model, x_np)


        # 预测类别

        # 用 argmax(logits) 即 最高分 对应的索引

        preds = np.argmax(logits.data, axis=1)


        # 正确个数

        correct += (preds == target.numpy()).sum()

        total += target.size(0)


    # 正确个数 / 批次个数

    return correct / total

    for epoch in range(num_epochs):

        # 每个 epoch 开始前清零模型参数的梯度

        model.zero_grad()

        optimizer.zero_grad()


        train_correct = 0

        train_total = 0


        for batch_idx, (data, target) in enumerate(train_loader):

            # 将 PyTorch 的 tensor 转为 numpy 数组并展开为向量

            x_np = data.view(data.shape[0], -1).numpy() # (batch, 784)

            y_np = target.numpy().astype(np.int64) # (batch,)


            # 前向传播

            # 得到 logits (Tensor)

            logits = forward_np_batch(model, x_np)


            # 计算损失

            loss = criterion(logits, y_np)


            # 反向传播

            # 将梯度累加到每个依赖的 Tensor.grad 上

            loss.backward()


            # 参数更新

            optimizer.step()


            # 清零梯度

            optimizer.zero_grad()


            # 统计训练集上的预测正确数(用于 epoch 结束后计算训练准确率)

            preds = np.argmax(logits.data, axis=1)

            train_correct += (preds == y_np).sum()

            train_total += y_np.shape[0]


        train_acc = train_correct / train_total

        test_acc = evaluate_accuracy(model, test_loader)


        print(f"Epoch {epoch+1}/{num_epochs}  train_acc: {train_acc:.4f}  test_acc: {test_acc:.4f}")

    import numpy as np

import torch

import torch.nn as nn

import torch.optim as optim

from torchvision import datasets, transforms

from torch.utils.data import DataLoader, Subset


class Tensor:

    """

    张量类,实现自动微分

    """

    def __init__(self, data, _parents = (), _op = '', label = '', requires_grad = True):

        # 确保 data 是 np 数组

        if not isinstance(data, np.ndarray):

            data = np.array(data, dtype=np.float64)


        self.data = data

        self.grad = np.zeros_like(self.data) # 梯度初始化为 0


        # 反向传播函数 

        self._backward = lambda: None


        # 记录当前 Tensor 的父节点

        self._prev = set(_parents)


        # 调试

        self._op = _op

        self.label = label


        # 是否需要计算梯度

        self.requires_grad = requires_grad


    def zero_grad(self):

        # 清零梯度

        self.grad = np.zeros_like(self.data)


    # 调试

    def __repr__(self):

        return f"Tensor(data = {self.data}, grad = {self.grad})"


    # 反向传播

    def backward(self):

        # 拓扑排序

        topo = []

        visited = set()

        # 由于是反向的,这里直接用dfs

        def build_topo(v):

            if v not in visited:

                visited.add(v)

                for parent in v._prev:

                    build_topo(parent)

                topo.append(v)


        build_topo(self)


        # 反向传播

        # 初始化,根节点求导是1

        self.grad = np.ones_like(self.data)


        # 反拓扑序

        for v in reversed(topo):

            v._backward()


    # 算术运算

    # 加

    def __add__(self, other):

        # 确保 other 是张量类

        other = other if isinstance(other, Tensor) else Tensor(other, requires_grad = False)

        # 创建 self 和 other 的儿子

        out = Tensor(self.data + other.data, (self, other), '+')


        # 反向传播函数

        def _backward():

            # a = b + c


            # dL/db = dL/da * da/db

            # d(b+c)/db = 1

            grad_self = 1 * out.grad

            # dL/dc = dL/da * da/dc

            # d(b+c)/dc = 1

            grad_other = 1* out.grad


            # 将梯度调整到正确的维度

            ndims_added_self = grad_self.ndim - self.data.ndim

            for i in range(ndims_added_self):

                # 将多出来的维度,沿着最外层的维度进行求和,来降维

                grad_self = grad_self.sum(axis=0)

            for i, dim in enumerate(self.data.shape):

                if dim == 1:

                    # 如果左操作数的某个维度尺寸为 1,要将最后的结果再保持维度,即从向量变为矩阵

                    # (3,) => (1, 3)

                    grad_self = grad_self.sum(axis=i, keepdims=True)


            ndims_added_other = grad_other.ndim - other.data.ndim

            for i in range(ndims_added_other):

                # 将多出来的维度,沿着最外层的维度进行求和,来降维

                grad_other = grad_other.sum(axis=0)

            for i, dim in enumerate(other.data.shape):

                if dim == 1:

                    # 如果左操作数的某个维度尺寸为 1,要将最后的结果再保持维度,即从向量变为矩阵

                    # (3,) => (1, 3)

                    grad_other = grad_other.sum(axis=i, keepdims=True)


            if self.requires_grad:

                self.grad += grad_self

            if other.requires_grad:

                other.grad += grad_other


        out._backward = _backward


        return out

    # 反向加

    def __radd__(self, other):

        return self + other


    # 取负

    def __neg__(self):

        return self * -1


    # 减

    def __sub__(self, other):

        # a = b - c = b + (-c)

        return self + (-other)

    # 反向减

    def __rsub__(self, other):

        return other + (-self)


    # 乘

    def __mul__(self, other):

        # 确保 other 是张量类

        other = other if isinstance(other, Tensor) else Tensor(other, requires_grad = False)

        # 创建 self 和 other 的儿子

        out = Tensor(self.data * other.data, (self, other), '*')


        # 反向传播函数

        def _backward():

            # a = b * c


            # dL/db = dL/da * da/db

            # d(a*b)/da = b

            grad_self = other.data * out.grad

            # dL/dc = dL/da * da/dc

            # d(a*b)/db = a

            grad_other = self.data * out.grad


            # 将梯度调整到正确的维度

            ndims_added_self = grad_self.ndim - self.data.ndim

            for i in range(ndims_added_self):

                # 将多出来的维度,沿着最外层的维度进行求和,来降维

                grad_self = grad_self.sum(axis=0)

            for i, dim in enumerate(self.data.shape):

                if dim == 1:

                    # 如果左操作数的某个维度尺寸为 1,要将最后的结果再保持维度,即从向量变为矩阵

                    # (3,) => (1, 3)

                    grad_self = grad_self.sum(axis=i, keepdims=True)


            ndims_added_other = grad_other.ndim - other.data.ndim

            for i in range(ndims_added_other):

                # 将多出来的维度,沿着最外层的维度进行求和,来降维

                grad_other = grad_other.sum(axis=0)

            for i, dim in enumerate(other.data.shape):

                if dim == 1:

                    # 如果左操作数的某个维度尺寸为 1,要将最后的结果再保持维度,即从向量变为矩阵

                    # (3,) => (1, 3)

                    grad_other = grad_other.sum(axis=i, keepdims=True)


            if self.requires_grad:

                self.grad += grad_self

            if other.requires_grad:

                other.grad += grad_other


        out._backward = _backward


        return out

    # 反向乘

    def __rmul__(self, other):

        return self * other


    # 除

    def __truediv__(self, other):

        return self * other**-1

    # 反向除

    def __rtruediv__(self, other):

        return other * self**-1


    # 幂运算

    def __pow__(self, other):

        # x^y

        other = other if isinstance(other, Tensor) else Tensor(other, requires_grad = False)

        out = Tensor(self.data ** other.data, (self, other), '**')


        def _backward():

            # d(x^y)/dx = y * x^(y-1)

            if np.all(self.data != 0):

                # 防止 0^负数

                grad_self = other.data * (self.data ** (other.data - 1)) * out.grad


                # 处理广播

                ndims_added = grad_self.ndim - self.data.ndim

                for i in range(ndims_added):

                    grad_self = grad_self.sum(axis=0)

                for i, dim in enumerate(self.data.shape):

                    if dim == 1:

                        grad_self = grad_self.sum(axis=i, keepdims=True)


                if self.requires_grad:

                    self.grad += grad_self


            # d(x^y)/∂y = x^y * ln(x)

            if np.all(self.data > 0):

                # 防止 ln(负数) 或 ln(0)

                grad_other = (self.data ** other.data) * np.log(self.data) * out.grad


                # 处理广播

                ndims_added = grad_other.ndim - other.data.ndim

                for i in range(ndims_added):

                    grad_other = grad_other.sum(axis=0)

                for i, dim in enumerate(other.data.shape):

                    if dim == 1:

                        grad_other = grad_other.sum(axis=i, keepdims=True)

                if other.requires_grad:

                    other.grad += grad_other


        out._backward = _backward


        return out

    # 反向幂运算,即指数函数

    def __rpow__(self, other):

        # a^x

        # other^self，other 是常数，self 是变量


        # 底数要大于0

        assert other > 0, "base must be positive for general exponential"


        out = Tensor(other ** self.data, (self,), f'{other}**x')


        def _backward():

            # d(a^x)/dx = a^x * ln(a)

            if self.requires_grad:

                self.grad += (other ** self.data) * np.log(other) * out.grad


        out._backward = _backward


        return out


    # e^x

    def exp(self):

        # e^x

        out = Tensor(np.exp(self.data), (self,), 'exp')


        def _backward():

            # d(e^x)/dx = e^x

            if self.requires_grad:

                self.grad += np.exp(self.data) * out.grad


        out._backward = _backward

        return out


    # log_a(x)

    def log_a(self, a):        

        # 底数和真数检查

        assert np.all(self.data > 0), f"log_{a} undefined for non-positive values"

        assert a > 0 and a != 1, "a must be positive and not equal to 1"


        out = Tensor(np.log(self.data) / np.log(a), (self,), f'log_{a}')


        def _backward():

            # d(log_a(x))/dx = 1 / (x * ln(a))

            if self.requires_grad:

                self.grad += (1.0 / (self.data * np.log(a))) * out.grad


        out._backward = _backward


        return out


    # ln

    def log(self):

        # ln(x)


        # 确保真数大于0

        assert np.all(self.data > 0), "log undefined for non-positive values"


        out = Tensor(np.log(self.data), (self,), 'log')


        def _backward():

            # d(ln(x))/dx = 1/x

            if self.requires_grad:

                self.grad += (1.0 / self.data) * out.grad


        out._backward = _backward

        return out


    # 三角函数

    # sin

    def sin(self):

        # sin(x)

        out = Tensor(np.sin(self.data), (self,), 'sin')


        def _backward():

            # d(sin(x))/dx = cos(x)

            if self.requires_grad:

                self.grad += np.cos(self.data) * out.grad


        out._backward = _backward


        return out


    # cos

    def cos(self):

        # cos(x)

        out = Tensor(np.cos(self.data), (self,), 'cos')


        def _backward():

            # d(cos(x))/dx = -sin(x)

            if self.requires_grad:

                self.grad += -np.sin(self.data) * out.grad


        out._backward = _backward


        return out


    # tan

    def tan(self):

        # tan(x)

        out = Tensor(np.tan(self.data), (self,), 'tan')


        def _backward():

            # d(tan(x))/dx = 1/((cos(x)^2)

            if self.requires_grad:

                self.grad += (1.0 / (np.cos(self.data) ** 2)) * out.grad


        out._backward = _backward


        return out


    # cot

    def cot(self):

        # cot(x)

        out = Tensor(1.0 / np.tan(self.data), (self,), 'cot')


        def _backward():

            # d(cot(x))/dx = -1/(sin(x)^2)

            if self.requires_grad:

                self.grad += (-1.0 / (np.sin(self.data) ** 2)) * out.grad


        out._backward = _backward


        return out


    # sec

    def sec(self):

        # sec(x)

        out = Tensor(1.0 / np.cos(self.data), (self,), 'sec')


        def _backward():

            # d(sec(x))/dx = tan(x)/cos(x)

            sec_val = 1.0 / np.cos(self.data)

            tan_val = np.tan(self.data)

            if self.requires_grad:

                self.grad += (sec_val * tan_val) * out.grad


        out._backward = _backward


        return out


    # csc

    def csc(self):

        # csc(x)

        out = Tensor(1.0 / np.sin(self.data), (self,), 'csc')


        def _backward():

            # d(sec(x))/dx = tan(x)/sin(x)

            csc_val = 1.0 / np.sin(self.data)

            cot_val = 1.0 / np.tan(self.data)

            if self.requires_grad:

                self.grad += (-csc_val * cot_val) * out.grad


        out._backward = _backward


        return out


    # 反三角函数

    # arcsin

    def arcsin(self):

        # arcsin(x)


        # 判断x是否在[-1,1]

        assert np.all(self.data >= -1) and np.all(self.data <= 1), "arcsin undefined outside [-1, 1]"

        out = Tensor(np.arcsin(self.data), (self,), 'arcsin')


        def _backward():

            # d(arcsin(x))/dx = 1/sqrt(1-x^2)

            if self.requires_grad:

                self.grad += (1.0 / np.sqrt(1 - self.data**2)) * out.grad


        out._backward = _backward


        return out


    # arccos

    def arccos(self):

        # arccos(x)


        # 判断x是否在[-1,1]

        assert np.all(self.data >= -1) and np.all(self.data <= 1), "arccos undefined outside [-1, 1]"


        out = Tensor(np.arccos(self.data), (self,), 'arccos')


        def _backward():

            # d(arccos(x))/dx = -1/sqrt(1-x^2)

            if self.requires_grad:

                self.grad += (-1.0 / np.sqrt(1 - self.data**2)) * out.grad


        out._backward = _backward


        return out


    # arctan

    def arctan(self):

        # arctan(x)

        out = Tensor(np.arctan(self.data), (self,), 'arctan')


        def _backward():

            # d(arctan(x))/dx = 1/1+x^2

            if self.requires_grad:

                self.grad += (1.0 / (1 + self.data**2)) * out.grad


        out._backward = _backward


        return out


    # 激活函数

    def relu(self):

        # relu: max(0,x)

        out = Tensor(np.maximum(0, self.data), (self,), 'relu')


        def _backward():

            # 导数: 1 if x > 0 else 0

            if self.requires_grad:

                self.grad += (self.data > 0) * out.grad


        out._backward = _backward


        return out


    def sigmoid(self):

        # sigmoid: f(x) = 1 / (1 + e^(-x))

        sigmoid_val = 1.0 / (1.0 + np.exp(-self.data))

        out = Tensor(sigmoid_val, (self,), 'sigmoid')


        def _backward():

            # sigmoid'(x) = sigmoid(x) * (1 - sigmoid(x))

            sig = 1.0 / (1.0 + np.exp(-self.data))

            if self.requires_grad:

                self.grad += sig * (1 - sig) * out.grad


        out._backward = _backward

        return out


    def softmax(self, axis=-1):

        # softmax(x_i) = e^(x_i) / Σ(e^(x_j))


        # 减去最大值

        x_shifted = self.data - np.max(self.data, axis=axis, keepdims=True)

        exp_x = np.exp(x_shifted)

        softmax_val = exp_x / np.sum(exp_x, axis=axis, keepdims=True)


        out = Tensor(softmax_val, (self,), 'softmax')


        def _backward():

            # ∂L/∂x = softmax(x) * (∂L/∂y - Σ(∂L/∂y * softmax(x)))

            s = softmax_val

            grad_input = s * (out.grad - np.sum(out.grad * s, axis=axis, keepdims=True))

            if self.requires_grad:

                self.grad += grad_input


        out._backward = _backward


        return out


    # 规约操作

    def sum(self, axis=None, keepdims=False):

        # sum

        out = Tensor(np.sum(self.data, axis=axis, keepdims=keepdims), (self,), 'sum')


        def _backward():

            # sum 的梯度: 广播回原始形状

            if axis is None:

                if self.requires_grad:

                    self.grad += np.ones_like(self.data) * out.grad

            else:

                # 需要恢复维度用于广播

                grad_expanded = np.expand_dims(out.grad, axis=axis)

                if self.requires_grad:

                    self.grad += np.ones_like(self.data) * grad_expanded


        out._backward = _backward


        return out


    def mean(self, axis=None, keepdims=False):

        # mean

        out = Tensor(np.mean(self.data, axis=axis, keepdims=keepdims), (self,), 'mean')


        def _backward():

            # mean 的梯度：1/n

            if axis is None:

                n = self.data.size

                if self.requires_grad:

                    self.grad += np.ones_like(self.data) * out.grad / n

            else:

                n = self.data.shape[axis]

                grad_expanded = np.expand_dims(out.grad, axis=axis)

                if self.requires_grad:

                    self.grad += np.ones_like(self.data) * grad_expanded / n


        out._backward = _backward


        return out


    # 形状操作

    def reshape(self, *shape):

        # reshape

        out = Tensor(self.data.reshape(shape), (self,), 'reshape')


        def _backward():

            # x 的梯度: y 的梯度的 reshape

            if self.requires_grad:

                self.grad += out.grad.reshape(self.data.shape)


        out._backward = _backward


        return out


    @property

    def T(self):

        # T

        # property 修饰是为了能 x.T 这样使用

        out = Tensor(self.data.T, (self,), 'transpose')


        def _backward():

            # x 的梯度: y 的梯度的 转置

            if self.requires_grad:

                self.grad += out.grad.T


        out._backward = _backward

        return out


    def matmul(self, other):

        # 矩阵乘法 @ 运算符

        other = other if isinstance(other, Tensor) else Tensor(other, requires_grad=False)

        out = Tensor(self.data @ other.data, (self, other), 'matmul')


        def _backward():

            # np.swapaxes是指定两个轴进行交换, 可以处理二维以上的情况

            # 这里是用来解决 批处理矩阵 的转置

            # 如(A, B, C)表示有 A 批, 每批是 B × C 的矩阵

            # 则(A, B, C)^T = (A, C, B)


            # d(AB)/dA = grad @ B^T

            if self.requires_grad:

                other_T = np.swapaxes(other.data, -1, -2)

                self.grad += out.grad @ other_T

            # d(AB)/dB = A^T @ grad

            if other.requires_grad:

                self_T = np.swapaxes(self.data, -1, -2)

                other.grad += self_T @ out.grad


        out._backward = _backward


        return out


    def __matmul__(self, other):

        return self.matmul(other)


# 损失函数

class Loss:

    # 损失函数基类

    def __call__(self, predictions, targets):

        # 计算损失值

        raise NotImplementedError


class MSELoss(Loss):

    # MSE

    # L = (1/n) * Σ(y_pred - y_true)^2

    def __call__(self, predictions, targets):

        # predictions: Tensor, 预测值

        # targets: Tensor, 真实值


        if not isinstance(targets, Tensor):

            targets = Tensor(targets, requires_grad=False)


        # MSE = mean((pred - target)^2)

        diff = predictions - targets

        squared_diff = diff ** 2

        loss = squared_diff.mean()


        return loss


class CrossEntropyLoss(Loss):

    # 交叉熵损失

    # L = -(1/n) * Σ y_true * log(y_pred)


    def __call__(self, logits, targets):

        # logits: Tensor, shape (batch_size, num_classes), 原始输出

        # targets: Tensor/np.ndarray, shape (batch_size,), 类别索引


        if not isinstance(targets, np.ndarray):

            targets = np.array(targets, dtype=np.int64)


        # 每批次样本大小

        batch_size = logits.data.shape[0]


        # 前向传播


        # log(P) = log(exp(logits) / sum(exp(logits))) = logits - log(sum(exp(logits)))


        # 减去最大值以防溢出

        max_logits = np.max(logits.data, axis=-1, keepdims=True)

        shifted_logits = logits.data - max_logits


        # LogSoftmax: log(P)

        log_softmax_data = shifted_logits - np.log(np.sum(np.exp(shifted_logits), axis=-1, keepdims=True))


        # 选择正确类别的概率

        # 创建 one-hot 编码

        targets_onehot = np.zeros_like(logits.data)

        targets_onehot[np.arange(batch_size), targets] = 1.0


        # 只提取正确类别的 log 概率

        correct_log_probs = log_softmax_data[np.arange(batch_size), targets]


        # loss = - (1/N) * sum(log(P[target]))

        loss_value = -np.sum(correct_log_probs) / batch_size


        # 反向传播


        # 新节点 记录 损失值

        out = Tensor(loss_value, (logits,), 'ce_loss') 


        def _backward():

            # dL/dz = P - Y_onehot (logits = z)


            # P

            P = np.exp(log_softmax_data) 


            # 梯度计算

            grad_logits = P - targets_onehot


            # 归一化

            grad_logits /= batch_size


            if logits.requires_grad:

                # 累加到 logits 的梯度

                logits.grad += grad_logits


        out._backward = _backward


        return out


# 神经网络层

class Module:

    # 神经网络模块基类

    def __call__(self, *args, **kwargs):

        return self.forward(*args, **kwargs)


    def forward(self, *args, **kwargs):

        # 前向传播

        raise NotImplementedError


    def parameters(self):

        # 返回模块的所有参数

        return []


    def zero_grad(self):

        # 清零所有参数的梯度

        for param in self.parameters():

            param.zero_grad()


class Linear(Module):

    # 全连接层(线性层)

    #  y = Wx + b


    def __init__(self, in_features, out_features, bias=True):

        # 初始化


        # in_features: 输入特征数

        # out_features: 输出特征数

        # bias: 是否使用偏置

        self.in_features = in_features

        self.out_features = out_features


        # Xavier 初始化

        limit = np.sqrt(6.0 / (in_features + out_features))

        self.weight = Tensor(

            np.random.uniform(-limit, limit, (in_features, out_features)),

            label='weight',

            requires_grad=True

        )


        if bias:

            self.bias = Tensor(

                np.zeros(out_features),

                label='bias',

                requires_grad=True

            )

        else:

            self.bias = None


    def forward(self, x):

        # 前向传播

        # x: shape (batch_size, in_features) 或 (in_features,)


        # y = Wx

        # @: 矩阵乘法

        out = x @ self.weight


        # y = y + b

        if self.bias is not None:

            out = out + self.bias


        return out


    def parameters(self):

        # 返回层的参数

        params = [self.weight]

        if self.bias is not None:

            params.append(self.bias)

        return params


    def __repr__(self):

        return f"Linear(in_features={self.in_features}, out_features={self.out_features}, bias={self.bias is not None})"


class Sequential(Module):

    # 顺序容器: 按顺序执行多个模块

    def __init__(self, *layers):

        self.layers = layers


    def forward(self, x):

        for layer in self.layers:

            x = layer(x)

        return x


    def parameters(self):

        # 返回所有层的参数

        params = []

        for layer in self.layers:

            params.extend(layer.parameters())

        return params


    def __repr__(self):

        layer_str = '\n  '.join([f"({i}): {layer}" for i, layer in enumerate(self.layers)])

        return f"Sequential(\n  {layer_str}\n)"


class ReLU(Module):

    # ReLU 激活函数层

    def forward(self, x):

        return x.relu()


class Sigmoid(Module):

    # Sigmoid 激活函数层

    def forward(self, x):

        return x.sigmoid()


class Softmax(Module):

    # Softmax 激活函数层

    def __init__(self, axis=-1):

        self.axis = axis


    def forward(self, x):

        return x.softmax(axis=self.axis)


# 优化器

class Optimizer:

    # 优化器基类

    def __init__(self, parameters, lr):

        self.parameters = parameters

        self.lr = lr


    def step(self):

        # 执行一步参数更新

        raise NotImplementedError


    def zero_grad(self):

        # 清零所有参数的梯度

        for param in self.parameters:

            param.zero_grad()


class SGD(Optimizer):

    # 随机梯度下降优化器

    def step(self):

        for param in self.parameters:

            if param.requires_grad:

                # θ = θ - lr * ∇θ

                param.data -= self.lr * param.grad


# 辅助函数

def forward_np_batch(model, x_np):

    # 将 numpy 的输入批次转为 Tensor（requires_grad=False）

    # 然后做前向传播。


    x_tensor = Tensor(x_np.astype(np.float64), requires_grad=False)

    out = model(x_tensor)

    return out


def evaluate_accuracy(model, data_loader):

    # 在给定的数据加载器上评估模型准确率

    correct = 0

    total = 0


    for data, target in data_loader:

        # 将 PyTorch 的 tensor 转为 numpy 数组并展开为向量

        x_np = data.view(data.shape[0], -1).numpy()


        # 原始输出

        logits = forward_np_batch(model, x_np)


        # 预测类别

        # 用 argmax(logits) 即 最高分 对应的索引

        preds = np.argmax(logits.data, axis=1)


        # 正确个数

        correct += (preds == target.numpy()).sum()

        total += target.size(0)


    # 正确个数 / 批次个数

    return correct / total


if __name__ == "__main__":

    # 测试

    # 训练参数

    batch_size = 64 # 训练批次大小

    test_batch_size = 1000 # 测试批次大小

    lr = 0.1 # 学习率

    num_epochs = 100 # 训练轮数

    hidden = 128 # 隐藏层神经元个数

    train_subset = None # 若为整数, 则只使用前 train_subset 个训练样本


    # 数据加载

    # torchvision 来 下载和加载 MNIST 数据集

    transform = transforms.Compose([transforms.ToTensor()])

    train_dataset = datasets.MNIST(root='./data', train=True, download=True, transform=transform)

    test_dataset = datasets.MNIST(root='./data', train=False, download=True, transform=transform)


    if train_subset is not None:

        train_dataset = Subset(train_dataset, list(range(train_subset)))


    train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)

    test_loader = DataLoader(test_dataset, batch_size=test_batch_size, shuffle=False)


    # 2 层 MLP

    model = Sequential(

        Linear(28*28, hidden),

        ReLU(),

        Linear(hidden, 10)

    )


    # 损失和优化器

    criterion = CrossEntropyLoss()

    optimizer = SGD(model.parameters(), lr=lr)


    # 训练

    for epoch in range(num_epochs):

        # 每个 epoch 开始前清零模型参数的梯度

        model.zero_grad()

        optimizer.zero_grad()


        train_correct = 0

        train_total = 0


        for batch_idx, (data, target) in enumerate(train_loader):

            # 将 PyTorch 的 tensor 转为 numpy 数组并展开为向量

            x_np = data.view(data.shape[0], -1).numpy() # (batch, 784)

            y_np = target.numpy().astype(np.int64) # (batch,)


            # 前向传播

            # 得到 logits (Tensor)

            logits = forward_np_batch(model, x_np)


            # 计算损失

            loss = criterion(logits, y_np)


            # 反向传播

            # 将梯度累加到每个依赖的 Tensor.grad 上

            loss.backward()


            # 参数更新

            optimizer.step()


            # 清零梯度

            optimizer.zero_grad()


            # 统计训练集上的预测正确数(用于 epoch 结束后计算训练准确率)

            preds = np.argmax(logits.data, axis=1)

            train_correct += (preds == y_np).sum()

            train_total += y_np.shape[0]


        train_acc = train_correct / train_total

        test_acc = evaluate_accuracy(model, test_loader)


        print(f"Epoch {epoch+1}/{num_epochs}\ntrain_acc: {train_acc:.4f}\ntest_acc: {test_acc:.4f}")
