strcpy, strncpy, memcpy, memmove 的 C++ 实现及 musl 源码分析

前言

最近对 复制函数 比较感兴趣，因为某次在写项目的时候遇到 memcpy 造成一些错误结果，比如下面的代码：

int arr[] = {1, 2, 3};
memcpy(arr + 1, arr, 2 * sizeof(int));

这里 dest 是 arr + 1，要复制 2 个 int，源区间 [arr, arr + 2) 和目标区间 [arr + 1, arr + 3) 重叠。memcpy 不保证 能正确处理重叠，很多实现是从前往后抄，最终会变成：

arr[] = {1, 1, 1}

也就是说，memcpy 的合法性要靠 调用者：区间不能重叠，dest 也要装得下 n 个字节，否则是 UB。重叠复制要用 memmove。

所以一不做二不休，我就去找了一些 复制函数 的 实现 及 musl 源码，才得出个所以然。

下面实现里会反复用到 字长（Word） 和 对齐掩码：

using Word = size_t;
constexpr size_t kAlignMask = sizeof(Word) - 1;

以及判断一个字里有没有 \0 字节：

static int word_has_zero(Word x) {
    const Word ones = ~Word(0) / 255u;   // 0x01010101...
    const Word highs = ones * 128u;      // 0x80808080...
    return ((x - ones) & ~x & highs) ? 1 : 0;
}

假设一个字是 4 字节：ones = 0x01010101，highs = 0x80808080。x - ones、~x、再和 highs 做与之后，某个字节是 0 时，对应的最高位会被置上，这样就能一次判断整字里有没有 \0。

要对齐再按字复制，是因为 dest 和 src 相对 sizeof(Word) 的余数要相同，再逐字节补齐到整字边界。有些架构访问不对齐地址会直接报错，就算不报错，硬件也往往要拆成多步。

strcpy

简述

这是个经典的 字符串复制函数，和 memcpy 一样，长度校验不在函数内部：

  • 它不会检查 目的缓冲区够不够，而是逐字节复制，遇到 \0 就停下（\0 也会写过去）
  • 所以 dest 至少要能放下 strlen(src) + 1 个字节，否则写出边界，是 UB
char dest[4] = "123";      // 只有 4 字节：'1','2','3','\0'
const char src[] = "12345";
strcpy(dest, src);         // 要写 6 字节，dest 装不下

实现

朴素实现就是逐字节抄到 \0：

char* strcpy(char* dest, const char* src) {
    if (dest == nullptr || src == nullptr) {
        return dest;
    }
    char* ret = dest;
    while ((*dest++ = *src++) != '\0') {
    }
    return ret;
}

按字优化时，先对齐，再整字复制，碰到带 \0 的字再改回逐字节：

char* strcpy(char* dest, const char* src) {
    if (dest == nullptr || src == nullptr) {
        return dest;
    }
    char* ret = dest;
    if ((reinterpret_cast<std::uintptr_t>(dest) & kAlignMask)
     == (reinterpret_cast<std::uintptr_t>(src) & kAlignMask)) {
        while ((reinterpret_cast<std::uintptr_t>(dest) & kAlignMask) != 0) {
            if ((*dest = *src) == '\0') {
                return ret;
            }
            ++dest;
            ++src;
        }
        auto* wd = reinterpret_cast<Word*>(dest);
        auto* ws = reinterpret_cast<const Word*>(src);
        while (!word_has_zero(*ws)) {
            *wd++ = *ws++;
        }
        dest = reinterpret_cast<char*>(wd);
        src = reinterpret_cast<const char*>(ws);
    }
    while ((*dest = *src) != '\0') {
        ++dest;
        ++src;
    }
    return ret;
}

源码

源码在 stpcpy.c

musl 里 strcpy 实际上调用的是 stpcpy：stpcpy 返回 结尾 \0 的位置，strcpy 再返回 dest

#include <string.h>
#include <stdint.h>
#include <limits.h>

#define ALIGN (sizeof(size_t))
#define ONES ((size_t)-1/UCHAR_MAX)
#define HIGHS (ONES * (UCHAR_MAX/2+1))
#define HASZERO(x) ((x)-ONES & ~(x) & HIGHS)

char *__stpcpy(char *restrict d, const char *restrict s)
{
#ifdef __GNUC__
	typedef size_t __attribute__((__may_alias__)) word;
	word *wd;
	const word *ws;
	if ((uintptr_t)s % ALIGN == (uintptr_t)d % ALIGN) {
		for (; (uintptr_t)s % ALIGN; s++, d++)
			if (!(*d=*s)) return d;
		wd=(void *)d; ws=(const void *)s;
		for (; !HASZERO(*ws); *wd++ = *ws++);
		d=(void *)wd; s=(const void *)ws;
	}
#endif
	for (; (*d=*s); s++, d++);

	return d;
}

weak_alias(__stpcpy, stpcpy);

strncpy

简述

和 strcpy 的区别是可以指定最多复制 n 个字节：

  • 若 src 的长度 小于 n，会把 dest 剩下的部分 补 \0
  • 若 src 的长度 大于等于 n，只会抄 n 个字节，不会 自动在末尾补 \0
const char src[] = "12345";  // 6 字节，含 '\0'
char dest[10];
strncpy(dest, src, sizeof(dest));
// dest 前 6 字节是 "12345\0"，后面 4 字节也是 '\0'

如果写成 strncpy(dest, src, 3)，dest 只有 '1','2','3'，没有 \0。

实现

和 strcpy 差不多，多了要处理 n，以及 src 提前结束时的补零：

char* strncpy(char* dest, const char* src, size_t n) {
    if (dest == nullptr || src == nullptr) {
        return dest;
    }
    char* ret = dest;
    if (dest == src || n == 0) {
        return ret;
    }
    if ((reinterpret_cast<std::uintptr_t>(dest) & kAlignMask)
     == (reinterpret_cast<std::uintptr_t>(src) & kAlignMask)) {
        while (n > 0 && (reinterpret_cast<std::uintptr_t>(dest) & kAlignMask) != 0) {
            if ((*dest = *src) == '\0') {
                while (n > 0) {
                    *dest++ = '\0';
                    --n;
                }
                return ret;
            }
            ++dest;
            ++src;
            --n;
        }
        auto* wd = reinterpret_cast<Word*>(dest);
        auto* ws = reinterpret_cast<const Word*>(src);
        while (n >= sizeof(Word) && !word_has_zero(*ws)) {
            *wd++ = *ws++;
            n -= sizeof(Word);
        }
        dest = reinterpret_cast<char*>(wd);
        src = reinterpret_cast<const char*>(ws);
    }
    while (n > 0 && (*dest = *src) != '\0') {
        ++dest;
        ++src;
        --n;
    }
    while (n > 0) {
        *dest++ = '\0';
        --n;
    }
    return ret;
}

#include <string.h>
#include <stdint.h>
#include <limits.h>

#define ALIGN (sizeof(size_t)-1)
#define ONES ((size_t)-1/UCHAR_MAX)
#define HIGHS (ONES * (UCHAR_MAX/2+1))
#define HASZERO(x) ((x)-ONES & ~(x) & HIGHS)

char *__stpncpy(char *restrict d, const char *restrict s, size_t n)
{
#ifdef __GNUC__
	typedef size_t __attribute__((__may_alias__)) word;
	word *wd;
	const word *ws;
	if (((uintptr_t)s & ALIGN) == ((uintptr_t)d & ALIGN)) {
		for (; ((uintptr_t)s & ALIGN) && n && (*d=*s); n--, s++, d++);
		if (!n || !*s) goto tail;
		wd=(void *)d; ws=(const void *)s;
		for (; n>=sizeof(size_t) && !HASZERO(*ws);
		       n-=sizeof(size_t), ws++, wd++) *wd = *ws;
		d=(void *)wd; s=(const void *)ws;
	}
#endif
	for (; n && (*d=*s); n--, s++, d++);
tail:
	memset(d, 0, n);
	return d;
}

weak_alias(__stpncpy, stpncpy);

memcpy

简述

memcpy 是 按字节块复制，不管里面是不是字符串，\0 也照样抄，不会遇到 \0 就停。

和前言里说的一样：源、目标区间 不能重叠，重叠是 UB。

实现

它的实现就很简单，还是一样先 逐字复制，再 逐字节复制：

void* memcpy(void* dest, const void* src, size_t n) {
    auto* d = static_cast<unsigned char*>(dest);
    auto* s = static_cast<const unsigned char*>(src);
    if (d == s || n == 0) {
        return dest;
    }
    if ((reinterpret_cast<std::uintptr_t>(d) & kAlignMask)
     == (reinterpret_cast<std::uintptr_t>(s) & kAlignMask)) {
        while (n > 0 && (reinterpret_cast<std::uintptr_t>(d) & kAlignMask) != 0) {
            *d++ = *s++;
            --n;
        }
        auto* wd = reinterpret_cast<Word*>(d);
        auto* ws = reinterpret_cast<const Word*>(s);
        while (n >= sizeof(Word)) {
            *wd++ = *ws++;
            n -= sizeof(Word);
        }
        d = reinterpret_cast<unsigned char*>(wd);
        s = reinterpret_cast<const unsigned char*>(ws);
    }
    while (n > 0) {
        *d++ = *s++;
        --n;
    }
    return dest;
}

musl 里更细：先把 src 对齐到 4 字节，再看 dest：

  • dest 也对齐了，就按 16 / 8 / 4 / 2 / 1 字节抄
  • dest 没对齐，先读出对齐的 w、x，再用移位拼出要对齐写进 dest 的 4 字节

小端下，dest 余 1 时可以写成 (w >> 24) | (x << 8)：w >> 24 取出 还没对齐写完的那一字节 放到新 4 字节的最低位，x << 8 把后面 3 字节接上去。musl 源码用 LS / RS 宏，大端、小端都能用。下面按 小端 写：

using U32 = std::uint32_t;

void* memcpy(void* dest, const void* src, size_t n) {
    auto* d = static_cast<unsigned char*>(dest);
    auto* s = static_cast<const unsigned char*>(src);

    while ((reinterpret_cast<std::uintptr_t>(s) % 4) != 0 && n > 0) {
        *d++ = *s++;
        --n;
    }

    if ((reinterpret_cast<std::uintptr_t>(d) % 4) == 0) {
        while (n >= 16) {
            *reinterpret_cast<U32*>(d + 0) = *reinterpret_cast<const U32*>(s + 0);
            *reinterpret_cast<U32*>(d + 4) = *reinterpret_cast<const U32*>(s + 4);
            *reinterpret_cast<U32*>(d + 8) = *reinterpret_cast<const U32*>(s + 8);
            *reinterpret_cast<U32*>(d + 12) = *reinterpret_cast<const U32*>(s + 12);
            s += 16;
            d += 16;
            n -= 16;
        }
        if ((n & 8) != 0) {
            *reinterpret_cast<U32*>(d + 0) = *reinterpret_cast<const U32*>(s + 0);
            *reinterpret_cast<U32*>(d + 4) = *reinterpret_cast<const U32*>(s + 4);
            d += 8;
            s += 8;
        }
        if ((n & 4) != 0) {
            *reinterpret_cast<U32*>(d + 0) = *reinterpret_cast<const U32*>(s + 0);
            d += 4;
            s += 4;
        }
        if ((n & 2) != 0) {
            *d++ = *s++;
            *d++ = *s++;
        }
        if ((n & 1) != 0) {
            *d = *s;
        }
        return dest;
    }

    if (n >= 32) {
        U32 w = 0;
        U32 x = 0;
        switch (reinterpret_cast<std::uintptr_t>(d) % 4) {
            case 1:
                w = *reinterpret_cast<const U32*>(s);
                *d++ = *s++;
                *d++ = *s++;
                *d++ = *s++;
                n -= 3;
                while (n >= 17) {
                    x = *reinterpret_cast<const U32*>(s + 1);
                    *reinterpret_cast<U32*>(d + 0) = (w >> 24) | (x << 8);
                    w = *reinterpret_cast<const U32*>(s + 5);
                    *reinterpret_cast<U32*>(d + 4) = (x >> 24) | (w << 8);
                    x = *reinterpret_cast<const U32*>(s + 9);
                    *reinterpret_cast<U32*>(d + 8) = (w >> 24) | (x << 8);
                    w = *reinterpret_cast<const U32*>(s + 13);
                    *reinterpret_cast<U32*>(d + 12) = (x >> 24) | (w << 8);
                    s += 16;
                    d += 16;
                    n -= 16;
                }
                break;
            case 2:
                w = *reinterpret_cast<const U32*>(s);
                *d++ = *s++;
                *d++ = *s++;
                n -= 2;
                while (n >= 18) {
                    x = *reinterpret_cast<const U32*>(s + 2);
                    *reinterpret_cast<U32*>(d + 0) = (w >> 16) | (x << 16);
                    w = *reinterpret_cast<const U32*>(s + 6);
                    *reinterpret_cast<U32*>(d + 4) = (x >> 16) | (w << 16);
                    x = *reinterpret_cast<const U32*>(s + 10);
                    *reinterpret_cast<U32*>(d + 8) = (w >> 16) | (x << 16);
                    w = *reinterpret_cast<const U32*>(s + 14);
                    *reinterpret_cast<U32*>(d + 12) = (x >> 16) | (w << 16);
                    s += 16;
                    d += 16;
                    n -= 16;
                }
                break;
            case 3:
                w = *reinterpret_cast<const U32*>(s);
                *d++ = *s++;
                n -= 1;
                while (n >= 19) {
                    x = *reinterpret_cast<const U32*>(s + 3);
                    *reinterpret_cast<U32*>(d + 0) = (w >> 8) | (x << 24);
                    w = *reinterpret_cast<const U32*>(s + 7);
                    *reinterpret_cast<U32*>(d + 4) = (x >> 8) | (w << 24);
                    x = *reinterpret_cast<const U32*>(s + 11);
                    *reinterpret_cast<U32*>(d + 8) = (w >> 8) | (x << 24);
                    w = *reinterpret_cast<const U32*>(s + 15);
                    *reinterpret_cast<U32*>(d + 12) = (x >> 8) | (w << 24);
                    s += 16;
                    d += 16;
                    n -= 16;
                }
                break;
            default:
                break;
        }
    }

    if ((n & 16) != 0) {
        for (int i = 0; i < 16; ++i) {
            *d++ = *s++;
        }
    }
    if ((n & 8) != 0) {
        for (int i = 0; i < 8; ++i) {
            *d++ = *s++;
        }
    }
    if ((n & 4) != 0) {
        for (int i = 0; i < 4; ++i) {
            *d++ = *s++;
        }
    }
    if ((n & 2) != 0) {
        *d++ = *s++;
        *d++ = *s++;
    }
    if ((n & 1) != 0) {
        *d = *s;
    }
    return dest;
}

memmove

简述

memmove 用来 安全处理重叠。思路是：

  • 若 dest 在 src 后面，且距离 < n，说明从前往后抄会覆盖还没读的源字节，这时要 从后往前 抄
  • 否则 从前往后 抄

实现

加个 简述 中说的 重叠部分 判断 就好了，然后写一下 从后往前赋值

void* memmove(void* dest, const void* src, size_t n) {
    auto* d = static_cast<unsigned char*>(dest);
    auto* s = static_cast<const unsigned char*>(src);
    if (d == s || n == 0) {
        return dest;
    }

    const auto ud = reinterpret_cast<std::uintptr_t>(d);
    const auto us = reinterpret_cast<std::uintptr_t>(s);
    if (ud > us && ud - us < n) {
        if (((us + n) & kAlignMask) == ((ud + n) & kAlignMask)) {
            while (((ud + n) & kAlignMask) != 0) {
                if (n == 0) {
                    return dest;
                }
                --n;
                d[n] = s[n];
            }
            while (n >= sizeof(Word)) {
                n -= sizeof(Word);
                *reinterpret_cast<Word*>(d + n) = *reinterpret_cast<const Word*>(s + n);
            }
        }
        while (n > 0) {
            --n;
            d[n] = s[n];
        }
        return dest;
    }
    if ((ud & kAlignMask) == (us & kAlignMask)) {
        while (n > 0 && (reinterpret_cast<std::uintptr_t>(d) & kAlignMask) != 0) {
            *d++ = *s++;
            --n;
        }
        auto* wd = reinterpret_cast<Word*>(d);
        auto* ws = reinterpret_cast<const Word*>(s);
        while (n >= sizeof(Word)) {
            *wd++ = *ws++;
            n -= sizeof(Word);
        }
        d = reinterpret_cast<unsigned char*>(wd);
        s = reinterpret_cast<const unsigned char*>(ws);
    }
    while (n > 0) {
        *d++ = *s++;
        --n;
    }
    return dest;
}

#ifdef __GNUC__
typedef __attribute__((__may_alias__)) size_t WT;
#define WS (sizeof(WT))
#endif

void *memmove(void *dest, const void *src, size_t n)
{
	char *d = dest;
	const char *s = src;

	if (d==s) return d;
	if ((uintptr_t)s-(uintptr_t)d-n <= -2*n) return memcpy(d, s, n);
