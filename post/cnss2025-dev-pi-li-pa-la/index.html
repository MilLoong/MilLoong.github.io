🎞️ 噼里啪啦动画

30% 在没有操作系统的虚拟机环境（或者真的硬件环境，如果你愿意）中，在屏幕上输出任何字符串，如Hello World!

  • 运行环境

    • 这题我依然直接用 WSL 来做了，然后是 WSL2
    • 虚拟机 用的是 Qemu（qemu-system-x86_64）
    • 启动方式 用的是 UEFI，因为我不是很想用 BIOS 来做
    • 指令集架构 是 x86
    • 编译工具： mingw-w64
    • UEFI库：gnu-efi
    • UEFI固件：OVMF.fd
    • 磁盘工具：mtools

  • 代码（./hello/hello.c）

    • #include <efi.h>

#include <efilib.h>


EFI_STATUS EFIAPI efi_main(EFI_HANDLE ImageHandle, EFI_SYSTEM_TABLE *SystemTable) {

    //初始化 GNU-EFI

    InitializeLib(ImageHandle, SystemTable);


    //清屏

    uefi_call_wrapper(SystemTable->ConOut->ClearScreen, 1, SystemTable->ConOut);


    //输出

    uefi_call_wrapper(SystemTable->ConOut->OutputString, 2, SystemTable->ConOut, L"Hello World!\r\n");


    //让程序一直运行

    while(1);

}

    • 这里我直接用了 gnu-efi 库，这是一个 C 语言 编写 UEFI 应用的开发库，可以省去写 EFI_SYSTEM_TABLE 这些结构体

  • 编译

    • 同上用 Makefile，其中：
      • make：编译
      • make run：运行

  • 运行

    • 由于我用的是 Qemu 运行，而 Qemu 不自带 UEFI 固件，所以先手动安装：

      $ sudo apt install qemu-system-x86 ovmf

    • Qemu 要启动 BOOTX64.EFI 文件，要将其放到 FAT驱动器分区，而 Qemu 刚好有将 目录 挂载为 虚拟FAT驱动器 的功能，就能将 某个 dir 目录（我这里叫 disk ）当作 FAT驱动器分区，然后就能启动了

      先：

      mkdir -p disk/EFI/BOOT

cp BOOTX64.EFI disk/EFI/BOOT/

      然后用：

      qemu-system-x86_64 -bios /usr/share/ovmf/OVMF.fd -drive file=fat:rw:disk,format=raw -net none

      启动

40% 基于上述要求，在屏幕上播放字符画或其他形式的 Bad Apple

  • 代码（./play_pilipala/play_pilipala.c）

    • #include <efi.h>

#include <efilib.h>


#define MAX_FRAME_SIZE (100 * 60) 


EFI_FILE_PROTOCOL *root_dir = NULL;


//打开根文件系统

EFI_STATUS open_root_dir(EFI_HANDLE ImageHandle) {

    EFI_STATUS Status;

    EFI_LOADED_IMAGE_PROTOCOL *LoadedImage;

    EFI_SIMPLE_FILE_SYSTEM_PROTOCOL *FileSystem;


    //获取 Loaded Image Protocol

    Status = uefi_call_wrapper(BS->HandleProtocol, 3, ImageHandle, &LoadedImageProtocol, (VOID**)&LoadedImage);

    if (EFI_ERROR(Status)) return Status;


    //获取 File System Protocol

    Status = uefi_call_wrapper(BS->HandleProtocol, 3, LoadedImage->DeviceHandle, &FileSystemProtocol, (VOID**)&FileSystem);

    if (EFI_ERROR(Status)) return Status;


    //打开根目录

    Status = uefi_call_wrapper(FileSystem->OpenVolume, 2, FileSystem, &root_dir);

    return Status;

}


//读取单个帧文件

EFI_STATUS read_frame(CHAR16 *filename, CHAR8 *buffer, UINTN buffer_size) {

    EFI_STATUS Status;

    EFI_FILE_PROTOCOL *file;

    UINTN read_size = buffer_size;


    //打开文件

    Status = uefi_call_wrapper(root_dir->Open, 5, root_dir, &file, filename, EFI_FILE_MODE_READ, 0);

    if (EFI_ERROR(Status)) return Status;


    //读取文件内容

    Status = uefi_call_wrapper(file->Read, 3, file, &read_size, buffer);


    //添加字符串结束符

    if (!EFI_ERROR(Status) && read_size < buffer_size) {

        buffer[read_size] = '\0';

    }


    //关闭文件

    uefi_call_wrapper(file->Close, 1, file);


    return Status;

}


//将 ASCII 字符串转换为宽字符并显示

void display_frame(EFI_SIMPLE_TEXT_OUT_PROTOCOL *ConOut, CHAR8 *ascii_frame) {

    //移动光标到左上角

    uefi_call_wrapper(ConOut->SetCursorPosition, 3, ConOut, 0, 0);


    //逐字符转换并显示

    CHAR16 wide_char[2] = {0, 0};

    for (UINTN i = 0; ascii_frame[i] != '\0'; i++) {

        if (ascii_frame[i] == '\n') {

            uefi_call_wrapper(ConOut->OutputString, 2, ConOut, L"\r\n");

        } else {

            wide_char[0] = (CHAR16)ascii_frame[i];

            uefi_call_wrapper(ConOut->OutputString, 2, ConOut, wide_char);

        }

    }

}


EFI_STATUS EFIAPI efi_main(EFI_HANDLE ImageHandle, EFI_SYSTEM_TABLE *SystemTable) {

    EFI_STATUS Status;

    InitializeLib(ImageHandle, SystemTable);


    EFI_SIMPLE_TEXT_OUT_PROTOCOL *ConOut = SystemTable->ConOut;


    //打开根文件系统

    Status = open_root_dir(ImageHandle);

    if (EFI_ERROR(Status)) {

        Print(L"Failed to open root directory: %r\n", Status);

        return Status;

    }


    //分配帧缓冲区

    CHAR8 *frame_buffer = AllocatePool(MAX_FRAME_SIZE);

    if (!frame_buffer) {

        Print(L"Failed to allocate memory\n");

        return EFI_OUT_OF_RESOURCES;

    }


    //清屏

    uefi_call_wrapper(ConOut->ClearScreen, 1, ConOut);

    uefi_call_wrapper(ConOut->EnableCursor, 2, ConOut, FALSE);  //隐藏光标


    //播放动画

    UINTN frame_num = 0;

    CHAR16 filename[256];


    while (1) {

        //构造文件名

        SPrint(filename, sizeof(filename), L"frames\\frame_%06d.txt", frame_num);


        //读取帧

        Status = read_frame(filename, frame_buffer, MAX_FRAME_SIZE - 1);

        if (EFI_ERROR(Status)) {

            //文件不存在，播放结束

            break;

        }


        //显示帧

        display_frame(ConOut, frame_buffer);


        //帧延时 微妙

        uefi_call_wrapper(BS->Stall, 1, 10000);


        frame_num++;

    }


    //播放完成

    FreePool(frame_buffer);

    uefi_call_wrapper(ConOut->EnableCursor, 2, ConOut, TRUE); //显示光标

    uefi_call_wrapper(ConOut->ClearScreen, 1, ConOut);

    Print(L"\nPlayback finished! Total frames: %d\n", frame_num - 1);


    //让程序一直运行

    while(1);


    return EFI_SUCCESS;

}

    • 这里就用一些有关的 api（打开文件什么的） 和 协议（Protocol） 来做就好了

    • 具体的实现逻辑就是，加载 frames 下的 .txt 文件，然后读取每个帧文件内容到缓冲区，经过一些转换字符等操作后，一帧一帧地播放

    • 带有字符画文件的 frames 文件夹要放到 ./play_pilipala/disk 下

  • 编译

    • 同上用 Makefile，其中：
      • make：编译
      • make run：运行

  • 运行

    • 同上

      先编译

      然后：

      mkdir -p disk/EFI/BOOT

cp BOOTX64.EFI disk/EFI/BOOT/

      再用：

      qemu-system-x86_64 -bios /usr/share/ovmf/OVMF.fd -drive file=fat:rw:disk,format=raw -net none

      启动

30% 基于上述要求，实现空格控制播放暂停和继续的功能

  • 代码（./play_pilipala_and_space_stop/play_pilipala_and_space_stop.c）

    • #include <efi.h>

#include <efilib.h>


#define MAX_FRAME_SIZE (100 * 60) //帧缓冲区大小


//根目录

EFI_FILE_PROTOCOL *root_dir = NULL;


//打开根文件系统

EFI_STATUS open_root_dir(EFI_HANDLE ImageHandle) {

    EFI_STATUS Status;

    EFI_LOADED_IMAGE_PROTOCOL *LoadedImage;

    EFI_SIMPLE_FILE_SYSTEM_PROTOCOL *FileSystem;


    //获取 Loaded Image Protocol

    Status = uefi_call_wrapper(BS->HandleProtocol, 3, ImageHandle, &LoadedImageProtocol, (VOID**)&LoadedImage);

    if (EFI_ERROR(Status)) return Status;


    //获取 File System Protocol

    Status = uefi_call_wrapper(BS->HandleProtocol, 3, LoadedImage->DeviceHandle, &FileSystemProtocol, (VOID**)&FileSystem);

    if (EFI_ERROR(Status)) return Status;


    //打开根目录

    Status = uefi_call_wrapper(FileSystem->OpenVolume, 2, FileSystem, &root_dir);

    return Status;

}


//读取单个帧文件

EFI_STATUS read_frame(CHAR16 *filename, CHAR8 *buffer, UINTN buffer_size) {

    EFI_STATUS Status;

    EFI_FILE_PROTOCOL *file;

    UINTN read_size = buffer_size;


    //打开文件

    Status = uefi_call_wrapper(root_dir->Open, 5, root_dir, &file, filename, EFI_FILE_MODE_READ, 0);

    if (EFI_ERROR(Status)) return Status;


    //读取文件内容

    Status = uefi_call_wrapper(file->Read, 3, file, &read_size, buffer);


    //添加字符串结束符

    if (!EFI_ERROR(Status) && read_size < buffer_size) {

        buffer[read_size] = '\0';

    }


    //关闭文件

    uefi_call_wrapper(file->Close, 1, file);


    return Status;

}


//将 ASCII 字符串转换为宽字符并显示

//显示一帧

void display_frame(EFI_SIMPLE_TEXT_OUT_PROTOCOL *ConOut, CHAR8 *ascii_frame) {

    //移动光标到左上角

    uefi_call_wrapper(ConOut->SetCursorPosition, 3, ConOut, 0, 0);


    //逐字符转换并显示

    CHAR16 wide_char[2] = {0, 0};

    for (UINTN i = 0; ascii_frame[i] != '\0'; i++) {

        if (ascii_frame[i] == '\n') {

            uefi_call_wrapper(ConOut->OutputString, 2, ConOut, L"\r\n");

        } else {

            wide_char[0] = (CHAR16)ascii_frame[i];

            uefi_call_wrapper(ConOut->OutputString, 2, ConOut, wide_char);

        }

    }

}


//显示暂停提示

void show_pause_hint(EFI_SIMPLE_TEXT_OUT_PROTOCOL *ConOut, BOOLEAN paused) {

    //在屏幕左上角显示状态

    uefi_call_wrapper(ConOut->SetCursorPosition, 3, ConOut, 0, 0);


    if (paused) {

        //暂停时显示黄色提示

        uefi_call_wrapper(ConOut->SetAttribute, 2, ConOut, EFI_TEXT_ATTR(EFI_YELLOW, EFI_BLACK));

        uefi_call_wrapper(ConOut->OutputString, 2, ConOut, L"Press SPACE to continue...");

    } else {

        //播放时清除提示(用空格覆盖)

        uefi_call_wrapper(ConOut->SetAttribute, 2, ConOut, EFI_TEXT_ATTR(EFI_LIGHTGRAY, EFI_BLACK));

        uefi_call_wrapper(ConOut->OutputString, 2, ConOut, L"                                            ");

    }


    //恢复默认颜色

    uefi_call_wrapper(ConOut->SetAttribute, 2, ConOut, EFI_TEXT_ATTR(EFI_LIGHTGRAY, EFI_BLACK));

}


EFI_STATUS EFIAPI efi_main(EFI_HANDLE ImageHandle, EFI_SYSTEM_TABLE *SystemTable) {

    EFI_STATUS Status;

    InitializeLib(ImageHandle, SystemTable);


    EFI_SIMPLE_TEXT_OUT_PROTOCOL *ConOut = SystemTable->ConOut;

    SIMPLE_INPUT_INTERFACE *ConIn = SystemTable->ConIn;  // 键盘输入接口


    //打开根文件系统

    Status = open_root_dir(ImageHandle);

    if (EFI_ERROR(Status)) {

        Print(L"Failed to open root directory: %r\n", Status);

        return Status;

    }


    //分配帧缓冲区

    CHAR8 *frame_buffer = AllocatePool(MAX_FRAME_SIZE);

    if (!frame_buffer) {

        Print(L"Failed to allocate memory\n");

        return EFI_OUT_OF_RESOURCES;

    }


    //清屏

    uefi_call_wrapper(ConOut->ClearScreen, 1, ConOut);

    uefi_call_wrapper(ConOut->EnableCursor, 2, ConOut, FALSE);  //隐藏光标


    //播放状态变量

    BOOLEAN paused = FALSE;  //暂停标志

    UINTN frame_num = 0;     //当前帧号

    CHAR16 filename[256];

    EFI_INPUT_KEY Key;       //按键结构


    //播放循环

    while (1) {

        //非阻塞式检查键盘输入

        //非阻塞式就是程序立即返回,不会卡住,然后循环检查输入

        Status = uefi_call_wrapper(ConIn->ReadKeyStroke, 2, ConIn, &Key);


        //如果有按键按下

        if (!EFI_ERROR(Status)) {

            if (Key.UnicodeChar == L' ') {

                //暂停播放

                //检查是否按下空格键

                paused = !paused;  //切换暂停状态

                show_pause_hint(ConOut, paused);

            } else if (Key.ScanCode == SCAN_ESC) {

                //退出播放

                //检查是否按下 ESC 键

                break;  //退出播放循环

            }

        }


        //如果处于暂停状态

        if (paused) {

            //短暂延时,降低 CPU 占用,然后继续检查键盘输入

            uefi_call_wrapper(BS->Stall, 1, 10000);  //10ms

            continue;  //跳过帧播放逻辑

        }


        //构造文件名

        SPrint(filename, sizeof(filename), L"frames\\frame_%06d.txt", frame_num);


        //读取帧

        Status = read_frame(filename, frame_buffer, MAX_FRAME_SIZE - 1);

        if (EFI_ERROR(Status)) {

            //文件不存在，播放结束

            break;

        }


        //显示帧

        display_frame(ConOut, frame_buffer);


        //帧延时 微秒

        //控制播放速度

        uefi_call_wrapper(BS->Stall, 1, 0);


        frame_num++;

    }


    //播放完成

    FreePool(frame_buffer);

    uefi_call_wrapper(ConOut->EnableCursor, 2, ConOut, TRUE); //显示光标

    uefi_call_wrapper(ConOut->ClearScreen, 1, ConOut);

    Print(L"\nPlayback finished! Total frames: %d\n", frame_num - 1);


    //让程序一直运行

    while(1);


    return EFI_SUCCESS;

}

    • 这里的话我好像没用到 时钟中断，就直接用 BS->Stall（熄火） 了，然后加个 paused 状态，每一帧检查一下（代码里改的是延迟检查）。不过感觉 播放性能 没什么影响，就 CPU 的利用率会有点影响，毕竟不用 时钟中断 的话 CPU 是一直运行的？不过这题感觉应该不用 时钟中断 也没什么影响，就先这样简单实现吧

+30% 屏幕上播放的同时，播放 Bad Apple 的音频；或者，播放春日影或 Ave Mujica

  • 思考

    • 播放音频

      • 格式

        • 要 播放音频，需要先把 .mp3 或 .wav 解码，转成 PCM 文件 badapple.raw

        • 那要生成上面的 badapple.raw，需要用到 ffmpeg，命令是：

          ffmpeg -i badapple.mp3 -f s16le -ac 1 -ar 44100 badapple.raw

      • 播放方式

        • 播放音频 需要 Qemu 有声卡，查了一下是有方法 模拟声卡 的。但我这里看了一下，好像 UEFI标准文档 里没有涉及和 播放音频 相关的 协议，如果要真的 播放音频 的话就要编写一个 驱动

          我搜了一下找到一个 AudioPkg 驱动 的项目，不过虽然这个能 播放音频，但要做到像题目说的那样 边播放字符画 边播放音频 就会有点困难，我的一个想法是启动 播放字符画 的程序时，启动这个驱动，不过好像有点无从下手

          那就只能 手写一个 声卡驱动 了，不过感觉写起来好像有点复杂（

          然后另一种方法是 蜂鸣器，但这是不是真的符合题目要求的 播放 Bad Apple 的音频 呢？毕竟 蜂鸣器 只能播放 单频率蜂鸣声 的旋律（不过我实测了一下 蜂鸣器 还是太难绷了，像不了一点）

          不过我也暂时没查到什么其他做法了，难道真要手写一个 声卡驱动？那感觉要估算一下 工作量 了，所以在这次提交先要个 hint，如果确定是写 声卡驱动 的话，那我再去学着写一下吧（）
